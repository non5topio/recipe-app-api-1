    """Test creating a user with email exceeding maximum length raises an error"""
    email = 'a' * 300 + '@example.com'
    password = 'ValidPass123'
    name = 'Valid Name'
    with self.assertRaises(ValueError) as context:
        get_user_model().objects.create_user(
            email=email,
            password=password,
            name=name,
        )
    self.assertIn('Email exceeds maximum length', str(context.exception))
